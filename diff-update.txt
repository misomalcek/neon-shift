'use client';
import { useMemo, useRef } from 'react';
import { useFrame } from '@react-three/fiber';
import * as THREE from 'three';
import { useGame, world } from '@/lib/game-store';
const COUNT = 42
const RANGE = 130 // drones live inside this radius of the player, wrapping around
const dummy = new THREE.Object3D()
const calm = new THREE.Color(' #00f0ff').multiplyScalar(1.6)
const angry = new THREE.Color(' #ff2040').multiplyScalar(2.4)
const mix = new THREE.Color()
type Drone = {
  pos: THREE.Vector3
  vel: THREE.Vector3
  agitation: number // 0 calm wanderer .. 1 hunter Ă˘â‚¬â€ť spreads between neighbors
}
export function Drones() {
  const mesh = useRef<THREE.InstancedMesh>(null)
  const setHeat = useGame((s) => s.setHeat)
  const loseShard = useGame((s) => s.loseShard)
  const hitCooldown = useRef(0)
  const elapsed = useRef(0)
  const drones = useMemo<Drone[]>(
    () =>
      Array.from({ length: COUNT }, (_, i) => ({
        pos: new THREE.Vector3(
          (Math.sin(i * 12.9898) * 0.5 + 0.5) * RANGE * 2 - RANGE,
          10 + (Math.sin(i * 78.233) * 0.5 + 0.5) * 35,
          (Math.sin(i * 39.425) * 0.5 + 0.5) * RANGE * 2 - RANGE
        ),
        vel: new THREE.Vector3(Math.sin(i), 0, Math.cos(i)).multiplyScalar(6),
        agitation: 0,
      })),
    []
  )
  const steer = useMemo(() => new THREE.Vector3(), [])
  const diff = useMemo(() => new THREE.Vector3(), [])
  const center = useMemo(() => new THREE.Vector3(), [])
  const avgVel = useMemo(() => new THREE.Vector3(), [])
  const playerV = useMemo(() => new THREE.Vector3(), [])
  useFrame((_, rawDt) => {
    const dt = Math.min(rawDt, 0.05)
    elapsed.current += dt
    const m = mesh.current
    if (!m) return
    const p = world.playerPos
    playerV.set(p.x, p.y, p.z)
    hitCooldown.current = Math.max(0, hitCooldown.current - dt)
    const shards = useGame.getState().shards
    // Carrying more shards makes you "louder" Ă˘â‚¬â€ť drones sense you from farther away.
    const senseRadius = 18 + Math.min(shards, 20) * 2.2
    let totalAgitation = 0
    for (let i = 0; i < COUNT; i++) {
      const d = drones[i]
      steer.set(0, 0, 0)
      center.set(0, 0, 0)
      avgVel.set(0, 0, 0)
      let neighbors = 0
      let neighborAgitation = 0
      for (let j = 0; j < COUNT; j++) {
        if (i === j) continue
        const o = drones[j]
        const distSq = d.pos.distanceToSquared(o.pos)
        if (distSq < 400) {
          neighbors++
          neighborAgitation = Math.max(neighborAgitation, o.agitation)
          center.add(o.pos)
          avgVel.add(o.vel)
          if (distSq < 36) {
            // separation
            diff.subVectors(d.pos, o.pos).normalize().multiplyScalar(14 / (Math.sqrt(distSq) + 0.1))
            steer.add(diff)
          }
        }
      }
      if (neighbors > 0) {
        center.divideScalar(neighbors)
        steer.add(diff.subVectors(center, d.pos).multiplyScalar(0.04)) // cohesion
        steer.add(avgVel.divideScalar(neighbors).sub(d.vel).multiplyScalar(0.08)) // alignment
      }
      // Sense the player
      const pd = d.pos.distanceTo(playerV)
      if (pd < senseRadius) {
        d.agitation = Math.min(1, d.agitation + dt * 1.2)
      } else {
        d.agitation = Math.max(0, d.agitation - dt * 0.15)
      }
      // Agitation is contagious Ă˘â‚¬â€ť panic ripples through the swarm (emergence)
      if (neighborAgitation > d.agitation) {
        d.agitation = Math.min(neighborAgitation, d.agitation + dt * 0.8)
      }
      // Hunters chase; calm drones wander on noise
      if (d.agitation > 0.4) {
        diff.subVectors(playerV, d.pos).normalize().multiplyScalar(10 * d.agitation)
        steer.add(diff)
      } else {
        const t = elapsed.current * 0.3 + i
        steer.x += Math.sin(t) * 1.4
        steer.z += Math.cos(t * 0.9) * 1.4
        steer.y += Math.sin(t * 0.6) * 0.6
      }
      // Altitude band
      if (d.pos.y < 6) steer.y += 6
      if (d.pos.y > 50) steer.y -= 6
      d.vel.addScaledVector(steer, dt)
      const maxSpeed = 8 + d.agitation * 15
      if (d.vel.length() > maxSpeed) d.vel.setLength(maxSpeed)
      d.pos.addScaledVector(d.vel, dt)
      // Wrap around the player so the swarm is infinite too
      if (d.pos.x - p.x > RANGE) d.pos.x -= RANGE * 2
      if (d.pos.x - p.x < -RANGE) d.pos.x += RANGE * 2
      if (d.pos.z - p.z > RANGE) d.pos.z -= RANGE * 2
      if (d.pos.z - p.z < -RANGE) d.pos.z += RANGE * 2
      // Contact: steal shards and scatter
      if (pd < 2.6 && hitCooldown.current === 0 && useGame.getState().started) {
        hitCooldown.current = 1.2
        loseShard()
        d.agitation = 0
        d.vel.set(Math.sin(i) * 20, 8, Math.cos(i) * 20)
      }
      totalAgitation += d.agitation
      dummy.position.copy(d.pos)
      dummy.lookAt(d.pos.x + d.vel.x, d.pos.y + d.vel.y, d.pos.z + d.vel.z)
      const s = 0.7 + d.agitation * 0.3
      dummy.scale.set(s, s, s * 1.6)
      dummy.updateMatrix()
      m.setMatrixAt(i, dummy.matrix)
      m.setColorAt(i, mix.lerpColors(calm, angry, d.agitation))
    }
    m.instanceMatrix.needsUpdate = true
    if (m.instanceColor) m.instanceColor.needsUpdate = true
    const heat = totalAgitation / COUNT
    if (Math.abs(useGame.getState().heat - heat) > 0.04) setHeat(heat)
  })
  return (
    <instancedMesh ref={mesh} args={[undefined, undefined, COUNT]} frustumCulled={false}>
      <octahedronGeometry args={[0.9, 0]} />
      <meshBasicMaterial toneMapped={false} />
    </instancedMesh>
  )
}

  useFrame((_, rawDt) => {
    const dt = Math.min(rawDt, 0.05)
    const g = group.current
    if (!g || !started) return
    const k = keys.current
    // Steering Ă˘â‚¬â€ť keyboard OR mobile joystick
    const kTurnInput = (k.KeyA || k.ArrowLeft ? 1 : 0) - (k.KeyD || k.ArrowRight ? 1 : 0)
    const kClimbInput = (k.KeyW || k.ArrowUp ? 1 : 0) - (k.KeyS || k.ArrowDown ? 1 : 0)
    const kBoosting = k.ShiftLeft || k.ShiftRight || k.Space
    // Combine keyboard + mobile (mobile joystick: x = steer left/right, y = climb up/down)
    const turnInput = kTurnInput !== 0 ? kTurnInput : -mobileInput.x
    const climbInput = kClimbInput !== 0 ? kClimbInput : -mobileInput.y

    world.heading += turnInput * TURN * dt

      }
      tm.instanceMatrix.needsUpdate = true
      if (tm.instanceColor) tm.instanceColor.needsUpdate = true
    }

    if (Math.abs(useGame.getState().speed - spd) > 4) setSpeed(spd)
    const cx = Math.floor(world.playerPos.x / CHUNK_SIZE)
  return (

      {/* Boost trail */}

      {/* Underside glow */}
      <pointLight position={[0, -1, 0]} color=" #ff2d95" intensity={6} distance={8} />
    </group>
  )

export type EnemyType = 'scout' | 'gunship' | 'tank' | 'turret'
export type EnemyConfig = {
  type: EnemyType
  count: number
  health: number
  speed: number
  damage: number
  fireRate: number // shots per second
  range: number
  score: number
}
export function getLevelEnemyConfig(level: number): EnemyConfig[] {
  const configs: EnemyConfig[] = []
  // Scouts Ă˘â‚¬â€ť fast, weak, always present
  configs.push({
    type: 'scout',
    count: Math.min(4 + level * 2, 18),
    health: 30 + level * 5,
    speed: 10 + level * 1.5,
    damage: 8 + level * 2,
    fireRate: 0.8 + level * 0.1,
    range: 40 + level * 3,
    score: 50,
  })
  // Gunships Ă˘â‚¬â€ť medium, appear from level 2
  if (level >= 2) {
    configs.push({
      type: 'gunship',
      count: Math.min(2 + level, 10),
      health: 60 + level * 10,
      speed: 7 + level,
      damage: 15 + level * 3,
      fireRate: 0.5 + level * 0.08,
      range: 55 + level * 4,
      score: 150,
    })
  }
  // Tanks Ă˘â‚¬â€ť slow, heavy, appear from level 3
  if (level >= 3) {
    configs.push({
      type: 'tank',
      count: Math.min(1 + Math.floor(level / 2), 6),
      health: 120 + level * 20,
      speed: 4 + level * 0.5,
      damage: 25 + level * 5,
      fireRate: 0.3 + level * 0.05,
      range: 65 + level * 5,
      score: 300,
    })
  }
  // Turrets Ă˘â‚¬â€ť stationary, appear from level 4
  if (level >= 4) {
    configs.push({
      type: 'turret',
      count: Math.min(Math.floor(level / 2), 8),
      health: 80 + level * 15,
      speed: 0,
      damage: 20 + level * 4,
      fireRate: 1.2 + level * 0.1,
      range: 80 + level * 6,
      score: 200,
    })
  }
  return configs
}

export const MED_CELL = 80
export function medPackAt(ix: number, iz: number, level: number): { x: number; y: number; z: number } | null {
  const r = hash2(ix, iz, 5050 + level)
  // More med packs on higher levels (harder)
  const threshold = 0.88 - level * 0.01
  if (r < threshold) return null
  const ox = hash2(ix, iz, 5151) - 0.5
  const oz = hash2(ix, iz, 5252) - 0.5
  const y = 10 + hash2(ix, iz, 5353) * 20
  return {
    x: (ix + 0.5 + ox * 0.7) * MED_CELL,
    y,
    z: (iz + 0.5 + oz * 0.7) * MED_CELL,
  }
}

export function getLevelExitPos(level: number): { x: number; y: number; z: number } {
  // Each level exit is further away
  const dist = 200 + level * 80
  const angle = (level * 1.618) * Math.PI * 2
  return {
    x: Math.sin(angle) * dist,
    y: 18,
    z: Math.cos(angle) * dist,
  }
